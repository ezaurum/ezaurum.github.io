void bubbleSort(int * array, int size)
{
	int temp = 0;
	int j=0;
	bool swapped = true;
	while(swapped)
	{
		j++;
		swapped = false;
	for( int i = 0; i < size -1 -j ; i++)
	{
		if ( array[i] > array[i+1] )
		{
			temp = array[i];
			array[i] = array[i+1];
			array[i+1] = temp;
			swapped = true;
		}
	}
j++;
	}
}

void swap(int * a, int * b)
{
	int temp = 0;
	temp = *a;
	*a = *b;
	*b = temp;
}

bool swapif(int * a, int * b)
{
	if ( *a > *b)
	{
		swap(a,b);
		return true;
	}
	else
	{
		return false;
	}
}

void bubbleSort2(int * array, int size)
{
	int j=0;
	bool swapped = true;
	while(swapped)
	{
	swapped=false;
	for( int i = 0; i < size -j -1 ; i++)
	{
		swapped = swapif(array+i,array+i+1) ? true : swapped;
	}
	j++;
	}
}
