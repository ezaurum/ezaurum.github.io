Selection Sort

2011년 1월 13일 글 
void selectSort(int * array, int size)
{
	int minPoint = 0;

	for( int j = 0; j &lt; size -1; j++)
	{
		minPoint = j;
		for (int i = j; i &lt; size ; i++)
			{
				if(array[i] &lt; array[minPoint])
				{
					minPoint = i;
				}
			}
			swap(array+j, array+minPoint);
	}

}
