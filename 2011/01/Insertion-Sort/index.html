void  insertionSort(int * array, int size)
{
	int j = 0;
	int temp = 0;
	for (int i = 1; i < size ; i++)	{
		j = 0;
		while(j array[j]) {
				j++;
			}
			else {
				temp = array[i];
				for ( int k = i; k > j; k--) {
					array[k] = array[k-1];
				}
				array[j] = temp;
				j = i;
			}
		}
	}
}
