2019년 5월 12일 레시피 
select b.*, a.* from log_table a,
somehow_table b where a.id = b.id
and check_result='Success' and a.column=123
INTO OUTFILE 'orders.csv' FIELDS TERMINATED BY ',' ENCLOSED BY '"' LINES TERMINATED BY '\n';
