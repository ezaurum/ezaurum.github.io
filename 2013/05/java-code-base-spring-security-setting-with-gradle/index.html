2013-05-28

java code base spring security setting github 리파지토리 build.gradle에 다음을 추가 repositories { mavenCentral() maven { url “http://repo.springsource.org/snapshot" } }

dependencies { def springSecurityVersion = “3.1.4.RELEASE” compile “org.springframework.security:spring-security-core:$springSecurityVersion” compile “org.springframework.security:spring-security-web:$springSecurityVersion” compile “org.springframework.security:spring-security-config:$springSecurityVersion” compile “org.springframework.security:spring-security-javaconfig:1.0.0.CI-SNAPSHOT” }
