준비물 - 7zip
%drive% – drive 문자
format %drive%: /FS:fat32 /V:Ubuntu8 /Q
7z x .iso -x!bin -x!programs -x!isolinuxisolinux.txt -x![BOOT] -o%drive%:
mkdir %drive%:syslinuxwin32
copy syslinuxwin32.* %drive%:syslinuxwin32
copy backg.png %drive%:
copy syslinux.cfg %drive%:
copy vesamenu.c32 %drive%:
copy makeboot.bat %drive%:
%drive%:
copy isolinux*.*
rd isolinux /s/q
del isolinux.bin
call %drive%:makeboot.bat
